SDK Examples

from openai import OpenAI

client = OpenAI(
    base_url="http://localhost:3000/v1",
    api_key="sk-router"  # or ROUTER_API_KEY
)

resp = client.chat.completions.create(
    model="router-auto",
    messages=[
      {"role": "system", "content": "You are helpful."},
      {"role": "user", "content": "Hello!"}
    ],
    temperature=0.7
)
print(resp.choices[0].message.content)
# Check routing: resp is from first healthy provider
import OpenAI from "openai";
const client = new OpenAI({
  baseURL: "http://localhost:3000/v1",
  apiKey: "sk-router",
});
const res = await client.chat.completions.create({
  model: "router-auto",
  messages: [{ role: "user", content: "Hello!" }],
});
console.log(res.choices[0].message.content);

Error Behavior

When all 3 providers fail, the router returns:

{
  "error": {
    "message": "All endpoints were tried and failed...",
    "type": "all_endpoints_failed",
    "code": "all_endpoints_failed",
    "attempted_providers": [
      "kilo-gateway",
      "opencode-zen",
      "nvidia-nim"
    ],
    "details": [ ...per-provider errors... ]
  }
}
